cmake_minimum_required(VERSION 3.22)
add_library(silverWaveBank STATIC WaveBank.cpp WaveBank.h)
target_compile_features(silverWaveBank PUBLIC cxx_std_17)
target_include_directories(silverWaveBank PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
juce_add_gui_app(silverWaveRomBuilder VERSION 0.1.0 COMPANY_NAME "NK Stuff"
    BUNDLE_ID "com.nkstuff.silverk.wavebuilder" PRODUCT_NAME "Silver-K Wave ROM Builder")
target_sources(silverWaveRomBuilder PRIVATE Main.cpp WaveEditor.cpp WaveEditor.h)
target_compile_definitions(silverWaveRomBuilder PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_REPORT_APP_USAGE=0)
target_link_libraries(silverWaveRomBuilder PRIVATE silverWaveBank juce::juce_audio_utils)
if(MSVC)
    target_compile_options(silverWaveRomBuilder PRIVATE /utf-8)
endif()
if(APPLE)
    target_sources(silverWaveRomBuilder PRIVATE "${ROOT}/LICENSE.md")
    set_source_files_properties("${ROOT}/LICENSE.md" PROPERTIES HEADER_FILE_ONLY TRUE MACOSX_PACKAGE_LOCATION Resources)
    add_custom_command(TARGET silverWaveRomBuilder POST_BUILD COMMAND /usr/bin/codesign --force --deep --sign - --timestamp=none "$<TARGET_BUNDLE_DIR:silverWaveRomBuilder>" VERBATIM)
endif()
add_executable(silverWaveBank_test WaveBank_test.cpp)
target_link_libraries(silverWaveBank_test PRIVATE silverWaveBank)
add_test(NAME silver_wave_bank COMMAND silverWaveBank_test)
