cmake_minimum_required(VERSION 3.22)
include("${CMAKE_CURRENT_LIST_DIR}/../../../../cmake/MacOSDefaults.cmake")
project(SilverWaveRomBuilder VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
enable_testing()
set(JUCE_SOURCE "" CACHE PATH "Optional pinned JUCE source checkout")
if(NOT JUCE_SOURCE)
    include(FetchContent)
    if(POLICY CMP0135)
        cmake_policy(SET CMP0135 NEW)
    endif()
    FetchContent_Declare(silver_juce
        URL https://codeload.github.com/dsp56300/JUCE/tar.gz/2a8699d723fe9279e1148275485ddb0e18957f56
        SOURCE_SUBDIR silver-download-only)
    FetchContent_MakeAvailable(silver_juce)
    set(JUCE_SOURCE "${silver_juce_SOURCE_DIR}")
endif()
include("${ROOT}/cmake/JuceCompatibility.cmake")
add_subdirectory("${JUCE_SOURCE}" juce EXCLUDE_FROM_ALL)
add_subdirectory(.. app)
