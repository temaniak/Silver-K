cmake_minimum_required(VERSION 3.22)
include(cmake/MacOSDefaults.cmake)
project(SilverK VERSION 0.4.4 LANGUAGES C CXX)
if(MSVC)

    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
endif()
enable_testing()

target_compile_features(silver_create_init_flash PRIVATE cxx_std_17)
add_subdirectory(source/novation/silverK/buildStandalone)
