if(KSTATION_STAGE1_REFERENCE)
  foreach(variant reference candidate)
    if(variant STREQUAL reference)
      set(stage_source "${KSTATION_STAGE1_REFERENCE}")
    else()
      set(stage_source "${K_SOURCE}")
    endif()
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/stage1-${variant}/kstationH8.cpp"
      COMMAND python3 "${CMAKE_CURRENT_SOURCE_DIR}/prepare_stage1_trace.py" "${stage_source}/kstationH8.cpp" "${CMAKE_CURRENT_BINARY_DIR}/stage1-${variant}/kstationH8.cpp"
      DEPENDS "${stage_source}/kstationH8.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_stage1_trace.py"
      VERBATIM)
    add_executable(kstation_stage1_${variant}_trace probe.cpp "${CMAKE_CURRENT_BINARY_DIR}/stage1-${variant}/kstationH8.cpp" "${stage_source}/kstationHardware.cpp" "${stage_source}/kstationDSP.cpp" "${SILVER_SOURCE}/source/framework/baseLib/sha256.cpp")
    target_compile_definitions(kstation_stage1_${variant}_trace PRIVATE K_RESEARCH_STAGE1_TRACE=1)
    target_link_libraries(kstation_stage1_${variant}_trace PRIVATE dsp56kEmu)
    target_include_directories(kstation_stage1_${variant}_trace PRIVATE "${stage_source}" "${SILVER_SOURCE}/source/cpu" "${SILVER_SOURCE}/source/framework")
    if(variant STREQUAL candidate)
      target_compile_definitions(kstation_stage1_${variant}_trace PRIVATE K_RESEARCH_STAGE1=1)
    endif()
    add_executable(kstation_stage1_${variant} probe.cpp "${stage_source}/kstationH8.cpp" "${stage_source}/kstationHardware.cpp" "${stage_source}/kstationDSP.cpp" "${SILVER_SOURCE}/source/framework/baseLib/sha256.cpp")
    target_link_libraries(kstation_stage1_${variant} PRIVATE dsp56kEmu)
    target_include_directories(kstation_stage1_${variant} PRIVATE "${stage_source}" "${SILVER_SOURCE}/source/cpu" "${SILVER_SOURCE}/source/framework")
    if(variant STREQUAL candidate)
      target_compile_definitions(kstation_stage1_${variant} PRIVATE K_RESEARCH_STAGE1=1)
    endif()
  endforeach()
  foreach(check NativeTimers HdiWait)
    add_executable(kstation${check}_test "${K_SOURCE}/kstation${check}_test.cpp")
    target_include_directories(kstation${check}_test PRIVATE "${K_SOURCE}" "${SILVER_SOURCE}/source/cpu")
  endforeach()
  add_executable(kstationStage1Guard_test "${K_SOURCE}/kstationStage1Guard_test.cpp" "${K_SOURCE}/kstationH8.cpp" "${K_SOURCE}/kstationHardware.cpp" "${K_SOURCE}/kstationDSP.cpp" "${SILVER_SOURCE}/source/framework/baseLib/sha256.cpp")
  target_link_libraries(kstationStage1Guard_test PRIVATE dsp56kEmu)
  target_include_directories(kstationStage1Guard_test PRIVATE "${K_SOURCE}" "${SILVER_SOURCE}/source/cpu" "${SILVER_SOURCE}/source/framework")
  set(SYNTHLIB_DEMO_MODE OFF)
  configure_file("${SILVER_SOURCE}/source/framework/synthLib/buildconfig.h.in" "${CMAKE_CURRENT_BINARY_DIR}/generated/buildconfig.h")
  add_library(kstation_stage1_device_compile OBJECT EXCLUDE_FROM_ALL "${K_SOURCE}/kstationDevice.cpp")
  target_link_libraries(kstation_stage1_device_compile PRIVATE dsp56kEmu)
  target_include_directories(kstation_stage1_device_compile PRIVATE "${K_SOURCE}" "${SILVER_SOURCE}/source/cpu" "${SILVER_SOURCE}/source/framework" "${CMAKE_CURRENT_BINARY_DIR}/generated")
endif()
