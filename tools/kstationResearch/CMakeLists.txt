cmake_minimum_required(VERSION 3.22)
project(KStationResearch LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ASMJIT_STATIC ON CACHE BOOL "" FORCE)
set(DSP56300_DEBUGGER OFF CACHE BOOL "" FORCE)
foreach(required SILVER_SOURCE DSP_SOURCE ASMJIT_SOURCE)
  if(NOT IS_DIRECTORY "${${required}}")
    message(FATAL_ERROR "Supply ${required} source directory")
  endif()
endforeach()
add_subdirectory("${ASMJIT_SOURCE}" asmjit EXCLUDE_FROM_ALL)
add_subdirectory("${DSP_SOURCE}/source/dsp56kBase" dspbase EXCLUDE_FROM_ALL)
add_subdirectory("${DSP_SOURCE}/source/dsp56kEmu" dspemu EXCLUDE_FROM_ALL)
if(APPLE)
  target_link_libraries(dsp56kBase PUBLIC "-framework CoreAudio" "-framework AudioToolbox")
endif()
set(K_SOURCE "${SILVER_SOURCE}/source/novation/kstation/kstationLib")
foreach(variant clean observe)
  if(variant STREQUAL observe)
    set(h8_source "${CMAKE_CURRENT_BINARY_DIR}/observe/kstationH8.cpp")
  else()
    set(h8_source "${K_SOURCE}/kstationH8.cpp")
  endif()
  add_executable(kstation_research_${variant} probe.cpp "${h8_source}" "${K_SOURCE}/kstationHardware.cpp" "${K_SOURCE}/kstationDSP.cpp")
  target_link_libraries(kstation_research_${variant} PRIVATE dsp56kEmu)
  target_include_directories(kstation_research_${variant} PRIVATE "${K_SOURCE}" "${SILVER_SOURCE}/source/cpu" "${SILVER_SOURCE}/source/framework")
  if(variant STREQUAL observe)
    target_compile_definitions(kstation_research_${variant} PRIVATE K_RESEARCH_OBSERVE=1)
  endif()
endforeach()
